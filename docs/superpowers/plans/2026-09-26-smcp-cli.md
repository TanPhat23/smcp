# smcp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `smcp`, a Bun-powered CLI tool that enables developers to bundle, redact secrets from, and share their Agent Skills and MCP server configurations across multiple AI agents (OpenCode, Claude Code, Claude Desktop, Cursor, Windsurf) via GitHub Gists and local packs.

**Architecture:** A modular CLI powered by Bun, Commander, and `@clack/prompts`. Declarative agent profiles map configurations without hardcoded tool logic; a deterministic secret redaction engine converts credentials to `${VAR}` placeholders; native `fetch` communicates with the GitHub Gist REST API using a user Personal Access Token (PAT); SHA-256 fingerprinting tracks version changes in a local history store (`~/.smcp/shares.json`).

**Tech Stack:** Bun (v1.3+), TypeScript, Commander, `@clack/prompts`, `picocolors`, `zod`, native `fetch`, `bun test`.

```typescript
import fs from "node:fs";
import path from "node:path";
import os from "node:os";
import type { AuthConfig, ShareHistory, ShareRecord } from "../types.ts";

const SMCP_DIR = path.join(os.homedir(), ".smcp");
const CONFIG_FILE = path.join(SMCP_DIR, "config.json");
const SHARES_FILE = path.join(SMCP_DIR, "shares.json");

function ensureSmcpDir() {
  if (!fs.existsSync(SMCP_DIR)) {
    fs.mkdirSync(SMCP_DIR, { recursive: true, mode: 0o700 });
  }
}

export function getAuthConfig(): AuthConfig {
  ensureSmcpDir();
  // Environment variable override
  if (process.env.GITHUB_TOKEN) {
    return { githubToken: process.env.GITHUB_TOKEN };
  }
  if (!fs.existsSync(CONFIG_FILE)) {
    return {};
  }
  try {
    return JSON.parse(fs.readFileSync(CONFIG_FILE, "utf8"));
  } catch {
    return {};
  }
}

export function saveAuthConfig(config: AuthConfig): void {
  ensureSmcpDir();
  fs.writeFileSync(CONFIG_FILE, JSON.stringify(config, null, 2), { mode: 0o600 });
}

export function clearAuthConfig(): void {
  if (fs.existsSync(CONFIG_FILE)) {
    fs.unlinkSync(CONFIG_FILE);
  }
}

export function getSharesHistory(): ShareHistory {
  ensureSmcpDir();
  if (!fs.existsSync(SHARES_FILE)) {
    return { shares: [] };
  }
  try {
    const raw = fs.readFileSync(SHARES_FILE, "utf8");
    return JSON.parse(raw);
  } catch {
    return { shares: [] };
  }
}

export function recordShare(record: ShareRecord): void {
  ensureSmcpDir();
  const history = getSharesHistory();
  const index = history.shares.findIndex((s) => s.name === record.name);
  if (index >= 0) {
    history.shares[index] = record;
  } else {
    history.shares.push(record);
  }
  fs.writeFileSync(SHARES_FILE, JSON.stringify(history, null, 2), "utf8");
}
```

```typescript
export interface GitHubGistFile {
  content: string;
  filename?: string;
}

export interface CreateGistPayload {
  description: string;
  public: boolean;
  files: Record<string, { content: string }>;
}

export class GitHubClient {
  private token: string;

  constructor(token: string) {
    this.token = token;
  }

  private headers() {
    return {
      Authorization: `Bearer ${this.token}`,
      Accept: "application/vnd.github+json",
      "User-Agent": "smcp-cli",
      "X-GitHub-Api-Version": "2022-11-28"
    };
  }

  async verifyUser(): Promise<{ login: string; name: string }> {
    const res = await fetch("https://api.github.com/user", {
      headers: this.headers()
    });
    if (!res.ok) {
      throw new Error(`GitHub Authentication failed: ${res.status} ${res.statusText}`);
    }
    return (await res.json()) as { login: string; name: string };
  }

  async createGist(payload: CreateGistPayload): Promise<{ id: string; html_url: string }> {
    const res = await fetch("https://api.github.com/gists", {
      method: "POST",
      headers: this.headers(),
      body: JSON.stringify(payload)
    });
    if (!res.ok) {
      const err = await res.text();
      throw new Error(`Failed to create Gist: ${res.status} ${err}`);
    }
    return (await res.json()) as { id: string; html_url: string };
  }

  async updateGist(gistId: string, payload: Partial<CreateGistPayload>): Promise<{ id: string; html_url: string }> {
    const res = await fetch(`https://api.github.com/gists/${gistId}`, {
      method: "PATCH",
      headers: this.headers(),
      body: JSON.stringify(payload)
    });
    if (!res.ok) {
      const err = await res.text();
      throw new Error(`Failed to update Gist ${gistId}: ${res.status} ${err}`);
    }
    return (await res.json()) as { id: string; html_url: string };
  }

  static async fetchGist(gistIdOrUrl: string, token?: string): Promise<{ files: Record<string, { content: string; filename: string }> }> {
    const id = gistIdOrUrl.replace(/^https:\/\/gist\.github\.com\/([^/]+\/)?/, "");
    const headers: Record<string, string> = {
      Accept: "application/vnd.github+json",
      "User-Agent": "smcp-cli"
    };
    if (token) {
      headers.Authorization = `Bearer ${token}`;
    }
    const res = await fetch(`https://api.github.com/gists/${id}`, { headers });
    if (!res.ok) {
      throw new Error(`Failed to fetch Gist ${id}: ${res.status} ${res.statusText}`);
    }
    return (await res.json()) as { files: Record<string, { content: string; filename: string }> };
  }
}
```

```typescript
import * as p from "@clack/prompts";
import pc from "picocolors";
import { getAuthConfig, saveAuthConfig, clearAuthConfig } from "../core/state.ts";
import { GitHubClient } from "../core/github.ts";

export async function authLoginCommand(): Promise<void> {
  p.intro(pc.bgCyan(pc.black(" smcp — GitHub Authentication ")));

  p.note(
    "To share packs as Gists, smcp requires a GitHub Personal Access Token (PAT).\n" +
    "Generate one here: https://github.com/settings/tokens/new?scopes=gist&description=smcp-cli",
    "Instructions"
  );

  const token = await p.password({
    message: "Enter your GitHub Personal Access Token (PAT):",
    validate: (val) => (!val ? "Token is required" : undefined)
  });

  if (p.isCancel(token)) {
    p.cancel("Authentication cancelled.");
    return;
  }

  const s = p.spinner();
  s.start("Verifying token with GitHub...");

  try {
    const client = new GitHubClient(token);
    const user = await client.verifyUser();
    saveAuthConfig({ githubToken: token, githubUser: user.login });
    s.stop(pc.green(`✔ Authenticated successfully as @${user.login}`));
    p.outro("Saved securely to ~/.smcp/config.json");
  } catch (err: any) {
    s.stop(pc.red("✖ Authentication failed"));
    p.cancel(err.message);
  }
}

export function authLogoutCommand(): void {
  clearAuthConfig();
  p.outro(pc.green("✔ Logged out. Token removed from ~/.smcp/config.json"));
}

export async function authStatusCommand(): Promise<void> {
  const config = getAuthConfig();
  if (!config.githubToken) {
    p.log.warn("Not logged in. Run: smcp auth login");
    return;
  }
  const s = p.spinner();
  s.start("Checking token validity...");
  try {
    const client = new GitHubClient(config.githubToken);
    const user = await client.verifyUser();
    s.stop(pc.green(`✔ Logged in as @${user.login}`));
  } catch (err: any) {
    s.stop(pc.red("✖ Stored token is invalid or expired."));
    p.log.message(`Error: ${err.message}`);
  }
}
```

```typescript
import * as p from "@clack/prompts";
import pc from "picocolors";
import fs from "node:fs";
import path from "node:path";
import { detectAgents, readInstalledMcpServers, scanSkills } from "../core/agents.ts";
import { redactMcpServers } from "../core/redactor.ts";
import { getAuthConfig, getSharesHistory, recordShare } from "../core/state.ts";
import { GitHubClient } from "../core/github.ts";
import { hashContent, hashObject } from "../utils/crypto.ts";
import { authLoginCommand } from "./auth.ts";
import type { Manifest, McpServerConfig, SkillEntry } from "../types.ts";

export async function shareCommand(options: { output?: string }): Promise<void> {
  p.intro(pc.bgCyan(pc.black(" smcp — Share Skills & MCPs ")));

  const agents = detectAgents();
  if (agents.length === 0) {
    p.cancel("No supported AI agents found on this machine.");
    return;
  }

  // 1. Collect all available MCP servers
  const availableServers: Record<string, McpServerConfig> = {};
  for (const agent of agents) {
    if (agent.mcpConfigPath) {
      const servers = readInstalledMcpServers(agent.mcpConfigPath);
      Object.assign(availableServers, servers);
    }
  }

  // 2. Collect all available skills
  const availableSkills: SkillEntry[] = [];
  for (const agent of agents) {
    if (agent.skillsDirPath) {
      const skills = scanSkills(agent.skillsDirPath);
      for (const s of skills) {
        if (!availableSkills.some((existing) => existing.name === s.name)) {
          availableSkills.push(s);
        }
      }
    }
  }

  const serverNames = Object.keys(availableServers);
  if (serverNames.length === 0 && availableSkills.length === 0) {
    p.cancel("No MCP servers or skills found to share.");
    return;
  }

  // 3. Prompt user selections
  let selectedServers: Record<string, McpServerConfig> = {};
  if (serverNames.length > 0) {
    const picked = await p.multiselect({
      message: "Select MCP Servers to include:",
      options: serverNames.map((name) => ({
        value: name,
        label: name,
        hint: availableServers[name].command || availableServers[name].url
      })),
      required: false
    });
    if (p.isCancel(picked)) return;
    for (const name of picked as string[]) {
      selectedServers[name] = availableServers[name];
    }
  }

  let selectedSkills: SkillEntry[] = [];
  if (availableSkills.length > 0) {
    const picked = await p.multiselect({
      message: "Select Skills to include:",
      options: availableSkills.map((sk) => ({
        value: sk.name,
        label: sk.name,
        hint: sk.path
      })),
      required: false
    });
    if (p.isCancel(picked)) return;
    selectedSkills = availableSkills.filter((sk) => (picked as string[]).includes(sk.name));
  }

  if (Object.keys(selectedServers).length === 0 && selectedSkills.length === 0) {
    p.cancel("No items selected.");
    return;
  }

  // 4. Redact secrets
  const sRedact = p.spinner();
  sRedact.start("Analyzing and redacting sensitive credentials...");
  const { redactedServers, requiredEnv } = redactMcpServers(selectedServers);
  sRedact.stop(pc.green(`✔ Redacted secrets. Generated ${requiredEnv.length} environment placeholders.`));

  // 5. Pack Details
  const packName = await p.text({
    message: "Pack name:",
    defaultValue: "my-agent-pack",
    validate: (val) => (!val || !/^[a-z0-9_-]+$/i.test(val) ? "Must be alphanumeric (hyphens/underscores allowed)" : undefined)
  });
  if (p.isCancel(packName)) return;

  const packDesc = await p.text({
    message: "Pack description:",
    defaultValue: "Shared Skills and MCP stack"
  });
  if (p.isCancel(packDesc)) return;

  // 6. Check existing shares & version bump
  const history = getSharesHistory();
  const existingShare = history.shares.find((s) => s.name === packName);
  let version = "1.0.0";
  let targetGistId: string | undefined = undefined;

  if (existingShare) {
    const [maj, min, pat] = existingShare.version.split(".").map(Number);
    const nextPatch = `${maj}.${min}.${pat + 1}`;

    const action = await p.select({
      message: `Pack '${packName}' was previously shared (${existingShare.version}). How to proceed?`,
      options: [
        { value: "update", label: `Update existing share (bumps to v${nextPatch})` },
        { value: "new", label: "Create new standalone share" }
      ]
    });
    if (p.isCancel(action)) return;

    if (action === "update") {
      version = nextPatch;
      targetGistId = existingShare.gistId;
    }
  }

  // 7. Bundle skill files
  const bundledSkills: SkillEntry[] = [];
  const gistFiles: Record<string, { content: string }> = {};

  for (const sk of selectedSkills) {
    const skillFiles: Record<string, string> = {};
    if (fs.statSync(sk.path).isDirectory()) {
      const allFiles = fs.readdirSync(sk.path, { recursive: true });
      for (const f of allFiles) {
        const full = path.join(sk.path, f as string);
        if (fs.statSync(full).isFile()) {
          const content = fs.readFileSync(full, "utf8");
          skillFiles[f as string] = content;
          gistFiles[`skills_${sk.name}_${(f as string).replace(/\//g, "_")}`] = { content };
        }
      }
    } else {
      const content = fs.readFileSync(sk.path, "utf8");
      skillFiles["SKILL.md"] = content;
      gistFiles[`skills_${sk.name}_SKILL.md`] = { content };
    }

    bundledSkills.push({
      name: sk.name,
      path: `skills/${sk.name}/SKILL.md`,
      description: sk.description,
      contentHash: hashObject(skillFiles),
      files: skillFiles
    });
  }

  const manifest: Manifest = {
    $schema: "https://smcp.dev/schema.json",
    name: packName,
    version,
    description: packDesc,
    createdAt: existingShare?.lastSharedAt || new Date().toISOString(),
    updatedAt: new Date().toISOString(),
    mcpServers: redactedServers,
    skills: bundledSkills,
    requiredEnv
  };

  gistFiles["smcp.json"] = { content: JSON.stringify(manifest, null, 2) };

  // 8. Publish or export locally
  if (options.output) {
    const outDir = path.resolve(options.output);
    fs.mkdirSync(outDir, { recursive: true });
    fs.writeFileSync(path.join(outDir, "smcp.json"), JSON.stringify(manifest, null, 2), "utf8");
    for (const sk of bundledSkills) {
      if (sk.files) {
        const skDir = path.join(outDir, "skills", sk.name);
        fs.mkdirSync(skDir, { recursive: true });
        for (const [fn, cnt] of Object.entries(sk.files)) {
          fs.writeFileSync(path.join(skDir, fn), cnt, "utf8");
        }
      }
    }
    p.outro(pc.green(`✔ Pack successfully exported to ${outDir}`));
    return;
  }

  // GitHub Gist Publishing
  let auth = getAuthConfig();
  if (!auth.githubToken) {
    await authLoginCommand();
    auth = getAuthConfig();
    if (!auth.githubToken) {
      p.cancel("Cannot publish without GitHub token.");
      return;
    }
  }

  const isPublic = await p.confirm({
    message: "Make Gist public? (No = secret unlisted Gist)",
    initialValue: true
  });
  if (p.isCancel(isPublic)) return;

  const sPub = p.spinner();
  sPub.start("Publishing pack to GitHub Gist...");

  try {
    const client = new GitHubClient(auth.githubToken);
    let resultUrl = "";
    let finalGistId = targetGistId;

    if (targetGistId) {
      const res = await client.updateGist(targetGistId, {
        description: `[smcp] ${packName} v${version} - ${packDesc}`,
        files: gistFiles
      });
      resultUrl = res.html_url;
    } else {
      const res = await client.createGist({
        description: `[smcp] ${packName} v${version} - ${packDesc}`,
        public: isPublic,
        files: gistFiles
      });
      resultUrl = res.html_url;
      finalGistId = res.id;
    }

    // Save record
    const serverFingerprints: Record<string, string> = {};
    for (const [k, v] of Object.entries(redactedServers)) {
      serverFingerprints[k] = hashObject(v);
    }
    const skillFingerprints: Record<string, string> = {};
    for (const sk of bundledSkills) {
      skillFingerprints[sk.name] = sk.contentHash || "";
    }

    recordShare({
      name: packName,
      version,
      targetType: "gist",
      targetUrl: resultUrl,
      gistId: finalGistId,
      lastSharedAt: new Date().toISOString(),
      fingerprints: {
        mcpServers: serverFingerprints,
        skills: skillFingerprints
      }
    });

    sPub.stop(pc.green(`✔ Pack published: ${pc.bold(resultUrl)}`));

    p.note(
      `Share this pack with anyone:\n  ${pc.cyan(`npx smcp install ${resultUrl}`)}`,
      "Share Command"
    );
    p.outro(pc.green("Done!"));
  } catch (err: any) {
    sPub.stop(pc.red("✖ Failed to publish pack"));
    p.cancel(err.message);
  }
}
```

```typescript
import * as p from "@clack/prompts";
import pc from "picocolors";
import fs from "node:fs";
import path from "node:path";
import { GitHubClient } from "../core/github.ts";
import { detectAgents, getAgentProfiles } from "../core/agents.ts";
import { mergeMcpServersIntoFile, installSkillFiles } from "../core/merger.ts";
import { ManifestSchema, type Manifest, type McpServerConfig } from "../types.ts";
import { expandHome } from "../utils/paths.ts";

export async function installCommand(source: string): Promise<void> {
  p.intro(pc.bgCyan(pc.black(" smcp — Install Agent Pack ")));

  const sFetch = p.spinner();
  sFetch.start(`Fetching pack from ${source}...`);

  let manifest: Manifest;
  let rawFiles: Record<string, string> = {};

  try {
    if (source.startsWith("http") && source.includes("gist.github.com")) {
      const gist = await GitHubClient.fetchGist(source);
      if (!gist.files["smcp.json"]) {
        throw new Error("Gist does not contain an smcp.json manifest file.");
      }
      manifest = ManifestSchema.parse(JSON.parse(gist.files["smcp.json"].content));
      for (const [filename, fileObj] of Object.entries(gist.files)) {
        rawFiles[filename] = fileObj.content;
      }
    } else if (fs.existsSync(source)) {
      const manifestPath = fs.statSync(source).isDirectory()
        ? path.join(source, "smcp.json")
        : source;
      const content = fs.readFileSync(manifestPath, "utf8");
      manifest = ManifestSchema.parse(JSON.parse(content));
    } else {
      throw new Error(`Unsupported source: ${source}. Must be a GitHub Gist URL or local path.`);
    }
    sFetch.stop(pc.green(`✔ Loaded pack: ${pc.bold(manifest.name)} (v${manifest.version})`));
  } catch (err: any) {
    sFetch.stop(pc.red("✖ Could not load pack"));
    p.cancel(err.message);
    return;
  }

  // Display summary
  const mcpNames = Object.keys(manifest.mcpServers || {});
  const skillNames = (manifest.skills || []).map((s) => s.name);
  console.log(`\n${pc.bold("Description:")} ${manifest.description || "(none)"}`);
  console.log(`${pc.bold("MCP Servers:")} ${mcpNames.join(", ") || "(none)"}`);
  console.log(`${pc.bold("Skills:")} ${skillNames.join(", ") || "(none)"}`);

  // Select target agents
  const detected = detectAgents();
  const allProfiles = getAgentProfiles();

  const targetAgentIds = await p.multiselect({
    message: "Select AI agent(s) to install into:",
    options: Object.entries(allProfiles).map(([id, prof]) => {
      const isDet = detected.some((d) => d.id === id);
      return {
        value: id,
        label: prof.name,
        hint: isDet ? "detected" : "configured"
      };
    }),
    required: true
  });
  if (p.isCancel(targetAgentIds)) return;

  // Handle Required Environment Variables
  const envValues: Record<string, string> = {};
  if (manifest.requiredEnv && manifest.requiredEnv.length > 0) {
    p.note("This pack requires environment variables for credentials/URLs.", "Environment Setup");
    for (const req of manifest.requiredEnv) {
      if (process.env[req.key]) {
        envValues[req.key] = process.env[req.key]!;
        p.log.info(`Using existing environment variable for ${pc.cyan(req.key)}`);
        continue;
      }

      const promptFn = req.isSecret ? p.password : p.text;
      const answer = await promptFn({
        message: `${req.key} (${req.description || "required value"}):`,
        validate: (val) => (!val ? `${req.key} is required` : undefined)
      });
      if (p.isCancel(answer)) return;
      envValues[req.key] = answer as string;
    }
  }

  // Resolve templated MCP servers
  const resolvedServers: Record<string, McpServerConfig> = {};
  for (const [sName, sConf] of Object.entries(manifest.mcpServers || {})) {
    const resolved: McpServerConfig = { ...sConf };

    if (sConf.env) {
      const updatedEnv: Record<string, string> = {};
      for (const [k, v] of Object.entries(sConf.env)) {
        updatedEnv[k] = v.replace(/\${([A-Z0-9_]+)}/g, (_, varName) => envValues[varName] || `\${${varName}}`);
      }
      resolved.env = updatedEnv;
    }

    if (sConf.args) {
      resolved.args = sConf.args.map((arg) =>
        arg.replace(/\${([A-Z0-9_]+)}/g, (_, varName) => envValues[varName] || `\${${varName}}`)
      );
    }

    resolvedServers[sName] = resolved;
  }

  // Perform Installation
  const sInst = p.spinner();
  sInst.start("Installing skills and configuring MCP servers...");

  for (const agentId of targetAgentIds as string[]) {
    const profile = allProfiles[agentId];
    if (!profile) continue;

    // Install MCP servers
    if (profile.mcpConfig && Object.keys(resolvedServers).length > 0) {
      const primaryPath = expandHome(profile.mcpConfig.paths[0]);
      mergeMcpServersIntoFile(primaryPath, resolvedServers, profile.mcpConfig.key || "mcpServers");
    }

    // Install Skills
    if (profile.skills && manifest.skills && manifest.skills.length > 0) {
      const primarySkillsDir = expandHome(profile.skills.paths[0]);

      for (const skill of manifest.skills) {
        const filesToInstall: Record<string, string> = {};

        if (skill.files) {
          Object.assign(filesToInstall, skill.files);
        } else {
          // Find matching raw files from Gist
          const prefix = `skills_${skill.name}_`;
          for (const [fname, content] of Object.entries(rawFiles)) {
            if (fname.startsWith(prefix)) {
              const relName = fname.replace(prefix, "");
              filesToInstall[relName] = content;
            }
          }
        }

        if (Object.keys(filesToInstall).length === 0) {
          filesToInstall["SKILL.md"] = `# ${skill.name}\n${skill.description || ""}\n`;
        }

        installSkillFiles(primarySkillsDir, skill.name, filesToInstall);
      }
    }
  }

  sInst.stop(pc.green("✔ Installation completed!"));
  p.outro(pc.green(`Pack '${manifest.name}' is now active in: ${(targetAgentIds as string[]).join(", ")}`));
}
```
